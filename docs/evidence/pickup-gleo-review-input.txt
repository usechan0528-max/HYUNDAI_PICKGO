// IN-PROCESS READINESS ONLY. No Android caller verification, receiver, or transport.
// Host authorization and session binding MUST NOT be populated from Intent extras.
import { randomUUID } from 'node:crypto';
import { performance } from 'node:perf_hooks';
import { Fault } from '../domain.mjs';

export const ACTIONS = Object.freeze(['GET_PICKUP_STATUS', 'REPORT_PICKUP_ARRIVAL']);
export const TICKET_TTL = 5000;
const plain = value => value !== null && typeof value === 'object' &&
  (Object.getPrototypeOf(value) === Object.prototype || Object.getPrototypeOf(value) === null);
const fail = (name, message = 'Pickup voice integration is unavailable.') =>
  ({ functionName: name, status: 'fail', message, mode: 'voice' });

// Local strict candidate contract, NOT a claim about actual Gleo Intent encoding.
// Only parameterless commands; JSON whitespace permitted, BOM/double encoding denied.
export function parseCommand(input) {
  if (!plain(input) || Object.keys(input).some(k => !['action', 'extras'].includes(k)) ||
      !ACTIONS.includes(input.action)) throw new TypeError('Unsupported command.');
  if (input.extras !== undefined) {
    if (!plain(input.extras)) throw new TypeError('Invalid extras.');
    const keys = Object.keys(input.extras);
    if (keys.length && (keys.length !== 1 || keys[0] !== 'ACTION_PARAM' ||
        typeof input.extras.ACTION_PARAM !== 'string' ||
        Buffer.byteLength(input.extras.ACTION_PARAM, 'utf8') > 1024 ||
        !/^[\x20\t\r\n]*\{[\x20\t\r\n]*\}[\x20\t\r\n]*$/.test(input.extras.ACTION_PARAM))) {
      throw new TypeError('Invalid parameters.');
    }
  }
  return input.action;
}

function publicResult(name, state) {
  const progress = { waiting: 'waiting_for_arrival', arrived: 'driver_reported_arrival', boarded: 'rider_confirmed_boarding' }[state.stage];
  if (!progress || typeof state.ready !== 'boolean' || typeof state.joined !== 'boolean') throw new Error('Invalid state.');
  return { functionName: name, status: 'success', mode: 'voice',
    message: name === ACTIONS[1] ? 'The driver has reported arrival at the agreed pickup point.' : 'The current pickup status is available.',
    data: JSON.stringify([{ progress, riderReady: state.ready, riderJoined: state.joined, locationChangePending: state.proposal !== null }]) };
}

/**
 * Trusted host port: attest(invocation, phase) -> fresh one-use object
 * { invocation, checkedAt: monotonic ms, caller: 'verified', ux: 'allowed', online: true }.
 * No default issuer exists. Tests supply explicit fake attestations; they are NOT
 * caller authentication or CarUxRestrictions evidence. Every phase needs a new grant.
 * Board must be the synchronous in-process Board, never an async HTTP client.
 */
export class PickupVoiceReadiness {
  #board; #attest; #now; #lastTime = -Infinity; #binding = null;
  #pending = null; #grants = new WeakSet(); #busy = false; #closed = false;
  constructor(board, { attest = () => null, now = () => performance.now() } = {}) {
    this.#board = board; this.#attest = attest; this.#now = now;
  }
  // Only a trusted host may call this; no credential is returned in a ticket.
  bind(session) {
    this.#pending = null; this.#binding = null;
    if (this.#closed) return;
    if (!plain(session) || typeof session.id !== 'string' || typeof session.token !== 'string' || !session.token) return;
    this.#binding = { id: session.id, token: session.token };
  }
  clear() { this.#pending = null; this.#binding = null; }
  dispose() { this.clear(); this.#closed = true; }
  #time() {
    const time = this.#now();
    if (!Number.isFinite(time) || time < this.#lastTime) throw new Error('Invalid clock.');
    this.#lastTime = time; return time;
  }
  #authorize(invocation, phase) {
    const grant = this.#attest(invocation, phase), time = this.#time();
    if (!plain(grant) || this.#grants.has(grant)) throw new Error('Unavailable.');
    this.#grants.add(grant);
    if (grant.invocation !== invocation || grant.caller !== 'verified' || grant.ux !== 'allowed' || grant.online !== true ||
        !Number.isFinite(grant.checkedAt) || time < grant.checkedAt || time - grant.checkedAt >= 1000) throw new Error('Unavailable.');
    return time;
  }
  #read() {
    if (!this.#binding) throw new Error('Unavailable.');
    const state = this.#board.read(this.#binding.id, this.#binding.token);
    if (state.role !== 'driver' || state.status !== 'active') throw new Error('Unavailable.');
    return state;
  }
  prepare(input) {
    const name = ACTIONS.includes(input?.action) ? input.action : 'UNKNOWN_ACTION';
    if (this.#busy || this.#closed) return { response: fail(name) };
    this.#busy = true;
    try {
      const invocation = Object.freeze({});
      const at = this.#authorize(invocation, 'prepare');
      // A new authenticated invocation invalidates the previous pending intent.
      this.#pending = null;
      let action;
      try { action = parseCommand(input); }
      catch { return { response: fail(name, 'This pickup voice command or its parameters are not supported.') }; }
      const state = this.#read();
      const ticket = Object.freeze({});
      this.#pending = { ticket, invocation, action, at, state, binding: this.#binding, requestId: randomUUID() };
      return { ticket };
    } catch { return { response: fail(name) }; }
    finally { this.#busy = false; }
  }
  dispatch(ticket) {
    const pending = this.#pending;
    const name = pending?.ticket === ticket ? pending.action : 'UNKNOWN_ACTION';
    if (this.#busy || this.#closed) return fail(name);
    this.#busy = true;
    try {
      // Consume before calling any external host port, including attest.
      this.#pending = null;
      if (!pending || pending.ticket !== ticket) return fail(name);
      const time = this.#authorize(pending.invocation, 'dispatch');
      if (this.#binding !== pending.binding || time - pending.at >= TICKET_TTL) return fail(name);
      const current = this.#read();
      if (pending.action === ACTIONS[0]) return publicResult(name, current);
      // No refresh-and-retry: capture the original version and lease, including place.
      if (current.version !== pending.state.version) return { functionName: name, status: 'asking',
        message: 'The pickup agreement changed. Check it and make a new arrival request.', mode: 'voice' };
      const result = this.#board.act(pending.binding.id, pending.binding.token, {
        type: 'arrive', requestId: pending.requestId, version: pending.state.version, view: pending.state.view
      });
      return publicResult(name, result);
    } catch (error) {
      return fail(name, error instanceof Fault && error.status === 409
        ? 'Arrival could not be recorded for the current pickup state. Check the agreement before trying again.'
        : 'Pickup voice integration is unavailable.');
    } finally { this.#busy = false; }
  }
}

### experiments/pickup-board/gleo/README.md
# Gleo 연동 준비 · 실행 경로 비활성

현재 결과는 **실제 Board와 연결해 시험한 Node 내부 어댑터**입니다. Gleo 자연어 호출, Android Intent 수신·응답, 호출자 인증, 차량 UX SDK 연결은 구현되지 않았습니다. 기존 HTTP 서버·웹 UI·WebView에 이 모듈을 연결하지 않았습니다. 기본 인스턴스는 모든 명령을 거부합니다.

## 사용과 재현

저장소 루트에서 `node --test experiments/pickup-board/*.test.mjs`를 실행합니다. `gleo.test.mjs`의 가짜 host attest만 시험용으로 사용합니다. 이 fixture를 실제 인증이나 정차 판정으로 복사하지 마세요.

`PickupVoiceReadiness(board, { attest, now })`가 동기 `Board`를 감쌉니다. `bind({ id, token })`은 신뢰할 호스트의 현재 운전자 세션 공급 지점입니다. Intent/음성 인자에서 세션이나 권한을 받으면 안 됩니다. 준비 단계에는 영속 세션 공급자가 없으며 자동 복원을 하지 않습니다.

1. 신뢰할 호스트가 검증한 세션으로 `bind`합니다. role 문자열은 신뢰하지 않고 실제 Board capability로 검증합니다.
2. `prepare({ action, extras })`는 `{ ticket }` 또는 `{ response }`를 반환합니다. ticket은 빈 객체의 동일성으로 식별하며 직렬화·복원할 수 없습니다. 새 인증된 명령은 이전 준비 요청을 폐기합니다.
3. `dispatch(ticket)`은 Callback 형태 객체를 반환합니다. 5초 이내 한 번만 사용하며 잘못된 ticket도 현재 대기 요청을 폐기합니다. 재시도는 사용자 새 요청으로만 시작합니다.
4. 배경 전환/세션 상실/사용자 전환에는 `clear()`, 수명주기 종료에는 `dispose()`를 호출해야 합니다. 다시 생성한 인스턴스는 바인딩·ticket이 없습니다. 이 lifecycle은 Node 시험이며 실제 Android 콜백 시험이 아닙니다.

`attest(invocation, phase)`는 prepare/dispatch 각각에 새 일회성 객체를 공급해야 합니다: `{ invocation, checkedAt, caller: 'verified', ux: 'allowed', online: true }`. `checkedAt`은 `now`와 같은 단조 시계이며 1초 미만만 허용합니다. 기본 issuer는 null을 반환합니다. 해당 인터페이스는 **인증 구현이 아닌 신뢰할 host 경계**입니다. 객체를 생성할 수 있다는 사실이 Gleo 발신자 검증을 의미하지 않습니다. 실제 공급자를 연결하기 전에 출처·권한·플랫폼 UX를 검증해야 합니다.

## 두 명령의 경계

| 이름 | 의미 | 인자 |
|---|---|---|
| GET_PICKUP_STATUS | 현재 유효 회차의 진행·준비 상태 조회 | 없음 |
| REPORT_PICKUP_ARRIVAL | 운전자의 명시적인 수동 도착 알림 | 없음 |

action은 대소문자·공백까지 정확히 일치해야 합니다. extras 생략, 빈 객체, 또는 `ACTION_PARAM` 단일 문자열의 빈 JSON 객체만 허용합니다. JSON 공백(space/tab/CR/LF)은 허용하고 BOM·중첩 인용·다른 타입·추가 key·1024바이트 초과는 거부합니다. 필수 인자가 없으므로 누락 질문은 발생하지 않습니다. 개별 extra 전달도 두 명령에서는 빈 extras일 때만 허용됩니다. 이는 문서 불일치 하의 로컬 후보 계약이지 Gleo 실제 전송형식 인증이 아닙니다.

도착은 prepare에서 확보한 version/view로만 수행합니다. 그사이 변경되면 asking 응답 후 새 요청을 요구합니다. 조회는 dispatch 시점의 최신 값을 읽습니다. 도착의 자체 view와 조회 view를 공유하지 않습니다. 탑승자 대행, 장소 변경/수락, 종료는 제공하지 않습니다. 기존 joined/waiting/no-proposal/역할/회차/버전/8초 서버 lease 검증을 그대로 적용합니다. 어댑터의 단조 5초 만료가 서버 Date.now 기반 lease보다 추가로 제한하며 기존 Board 시계 정책을 변경하지 않습니다.

응답은 success/asking/fail, 영어 message, mode=voice를 사용합니다. data는 객체 배열의 JSON 문자열이며 progress/riderReady/riderJoined/locationChangePending만 포함합니다. 위치·세션·초대·view·요청ID·사건 원문은 반환하거나 로그하지 않습니다. 위치를 자동 판정하거나 상대 탑승을 대신 확인하지 않습니다. 알 수 없는 action은 로컬 오류의 `UNKNOWN_ACTION`으로 표시하며 Android로 방송하는 경로는 없습니다.

## 실제 연결 전에 필요한 것

- 제품 Android 패키지·버전을 확정하고 안전한 native 운전자 세션 공급자를 마련합니다. `org.example.pickupboard`/`0.0.0`은 미등록 후보 값입니다. WebView sessionStorage를 JavaScript bridge로 꺼내는 방식은 구현하지 않았습니다.
- 공식 Gleo 호출자 및 callback 수신 대상의 package/signature/permission 계약을 확보합니다. action·packageName extra·schema 등록 자체는 인증 증거로 사용하지 않습니다.
- 실제 수신 Intent에서 인자 형식을 검증하고 후보 parser 허용 규칙과 대조합니다. 문서의 개별 putExtra와 ACTION_PARAM 설명 불일치는 미해소입니다.
- 검증된 Android 수명주기/UX/온라인 상태를 host 포트에 연결합니다. default deny를 true 상수로 교체하지 않습니다. 비동기·HTTP Board로 교체하면 현재 동기 원자성 증거가 무효이며 취소·timeout·중복 실행을 다시 설계/검증해야 합니다.
- `contract/schema.candidate.json`을 검토 후 제품의 `assets/schema.json`으로 배치하고 action filter를 연결합니다. 현재 APK에는 후보 파일이 들어가지 않습니다. 외부 제출·권한 변경은 별도 사용자 승인 대상입니다.
- 공식 callback action `ai.pleos.playground.intent.action.APP_ACTION_RESPONSE`, extras `packageName`과 `response` JSON 문자열 계약에 맞춰 **확인된 수신 대상**으로 송신하는 transport를 구현합니다. 지금은 callback 객체 생성만 검증했습니다. 7초 플랫폼 timeout은 side effect rollback 보장이 아닙니다.
- 스키마 검토·승인 후 실제 Gleo 자연어→새 Intent/재진입→앱 상태 변화→실제 callback 수신까지 시험합니다. Android 빌드만으로 이 게이트를 통과하지 않습니다.

import { randomBytes, randomUUID } from 'node:crypto';
export const PLACES = ['가상 문화센터 정문', '가상 문화센터 동쪽 입구', '가상 공원 안내소'];
export const VIEW_TTL = 8000;
const token = () => randomBytes(24).toString('base64url');
export class Fault extends Error {
  constructor(status, message) { super(message); this.status = status; }
}
const requireThat = (ok, status, message) => { if (!ok) throw new Fault(status, message); };
export class Board {
  constructor(now = Date.now) { this.now = now; this.rooms = new Map(); }
  prune() { for (const [id, room] of this.rooms) if (this.now() - room.createdAt > 86400000) this.rooms.delete(id); }
  create(place) {
    this.prune();
    requireThat(PLACES.includes(place), 400, '목록에서 승차지점을 선택해 주세요.');
    requireThat(this.rooms.size < 100, 429, '회차가 너무 많습니다. 로컬 서버를 재시작해 주세요.');
    const room = { id: randomUUID(), createdAt: this.now(), version: 1, status: 'active', place,
      ready: false, joined: false, stage: 'waiting', outcome: null, proposal: null, lastEvent: '운전자가 약속을 만들었습니다.',
      driver: token(), rider: null, invite: token(), views: new Map(), requests: new Set() };
    this.rooms.set(room.id, room);
    return { id: room.id, token: room.driver, invite: room.invite, role: 'driver' };
  }
  room(id) { this.prune(); const r = this.rooms.get(id); requireThat(r, 404, '회차가 없거나 만료되었습니다. 서버 재시작 시 회차가 사라집니다.'); return r; }
  join(id, invite) {
    const r = this.room(id);
    requireThat(r.status === 'active' && !r.joined && typeof invite === 'string' && invite === r.invite, 403, '초대가 유효하지 않거나 이미 사용되었습니다.');
    r.rider = token(); r.invite = null; r.joined = true; r.version++; r.lastEvent = '탑승자가 참여했습니다.';
    return { id, token: r.rider, role: 'rider' };
  }
  auth(r, credential) {
    requireThat(typeof credential === 'string' && credential.length > 0, 401, '참여 권한이 없습니다.');
    const role = credential === r.driver ? 'driver' : credential === r.rider ? 'rider' : null;
    requireThat(role, 403, '이 회차의 참여자가 아닙니다.'); return role;
  }
  read(id, credential) {
    const r = this.room(id), role = this.auth(r, credential), view = token();
    // A short bounded list tolerates polling and command requests crossing in flight.
    const views = (r.views.get(credential) || []).filter(v => this.now() - v.at < VIEW_TTL).slice(-9);
    views.push({ key: view, at: this.now() }); r.views.set(credential, views);
    return { id, role, version: r.version, status: r.status, place: r.place, ready: r.ready,
      joined: r.joined, stage: r.stage, outcome: r.outcome, proposal: r.proposal ? { ...r.proposal } : null, lastEvent: r.lastEvent, view, ttl: VIEW_TTL };
  }
  act(id, credential, command) {
    const r = this.room(id), role = this.auth(r, credential);
    requireThat(command && typeof command === 'object', 400, '요청 형식이 올바르지 않습니다.');
    const { requestId, version, view, type } = command;
    requireThat(typeof requestId === 'string' && /^[a-zA-Z0-9-]{8,80}$/.test(requestId), 400, '요청 식별자가 필요합니다.');
    requireThat(!r.requests.has(requestId), 409, '이미 처리된 요청입니다. 최신 상태를 확인해 주세요.');
    requireThat(r.status === 'active', 409, '종료된 회차는 바꿀 수 없습니다.');
    requireThat(r.version === version, 409, '다른 변경이 있습니다. 최신 상태를 확인하고 다시 선택해 주세요.');
    requireThat((r.views.get(credential) || []).some(v => v.key === view && this.now() - v.at < VIEW_TTL), 409, '조회한 상태가 오래되었습니다. 새로 확인해 주세요.');
    requireThat(r.requests.size < 1000, 429, '회차의 변경 한도를 초과했습니다.');
    if (['ready', 'propose', 'accept'].includes(type)) requireThat(r.stage !== 'boarded', 409, '탑승 확인 후에는 준비나 장소를 바꿀 수 없습니다. 잘못 눌렀다면 탑승 확인을 취소해 주세요.');
    if (type === 'arrive') {
      requireThat(role === 'driver', 403, '운전자만 도착을 알릴 수 있습니다.');
      requireThat(r.joined && r.stage === 'waiting' && !r.proposal, 409, '상대 참여와 장소 합의를 확인해 주세요. 도착은 한 번만 알릴 수 있습니다.');
      r.stage = 'arrived'; r.lastEvent = '운전자가 확정 승차지점에 도착했다고 알렸습니다.';
    } else if (type === 'board') {
      requireThat(role === 'rider', 403, '탑승자만 탑승을 확인할 수 있습니다.');
      requireThat(r.stage === 'arrived' && !r.proposal, 409, '운전자 도착과 장소 합의를 먼저 확인해 주세요.');
      r.stage = 'boarded'; r.lastEvent = '탑승자가 탑승했다고 확인했습니다. 약속 종료를 기다립니다.';
    } else if (type === 'undoArrival') {
      requireThat(role === 'driver', 403, '운전자만 도착 표시를 취소할 수 있습니다.');
      requireThat(r.stage === 'arrived', 409, '탑승 확인 전 도착 표시만 취소할 수 있습니다.');
      r.stage = 'waiting'; r.lastEvent = '운전자가 도착 표시를 취소했습니다.';
    } else if (type === 'undoBoarding') {
      requireThat(role === 'rider', 403, '탑승자만 탑승 확인을 취소할 수 있습니다.');
      requireThat(r.stage === 'boarded', 409, '탑승 확인 상태가 아닙니다.');
      r.stage = 'arrived'; r.lastEvent = '탑승자가 탑승 확인을 취소했습니다. 운전자 도착 표시는 유지됩니다.';
    } else if (type === 'ready') {
      requireThat(role === 'rider', 403, '탑승자만 준비 상태를 바꿀 수 있습니다.');
      requireThat(typeof command.ready === 'boolean', 400, '준비 상태는 참 또는 거짓이어야 합니다.');
      r.ready = command.ready; r.lastEvent = r.ready ? '탑승자가 준비되었습니다.' : '탑승자가 아직 준비 중입니다.';
    } else if (type === 'propose') {
      requireThat(r.joined && !r.proposal, 409, '상대 참여 또는 기존 제안 처리를 기다려 주세요.');
      requireThat(PLACES.includes(command.place) && command.place !== r.place, 400, '현재 지점과 다른 지점을 선택해 주세요.');
      r.proposal = { id: randomUUID(), by: role, place: command.place };
      r.lastEvent = '장소 변경을 제안했습니다. 아직 확정되지 않았습니다.';
    } else if (type === 'accept' || type === 'reject') {
      requireThat(r.proposal && r.proposal.id === command.proposalId, 409, '현재 제안과 일치하지 않습니다.');
      requireThat(r.proposal.by !== role, 403, '자신의 제안을 직접 수락하거나 거절할 수 없습니다.');
      const resetArrival = type === 'accept' && r.stage === 'arrived';
      if (type === 'accept') { r.place = r.proposal.place; r.stage = 'waiting'; }
      r.proposal = null; r.lastEvent = type === 'accept' ? '변경이 수락되어 승차지점을 변경했습니다.' : '변경이 거절되어 기존 승차지점을 유지합니다.';
      if (resetArrival) r.lastEvent += ' 이전 도착 표시는 취소되었습니다. 새 장소에서 다시 알려주세요.';
    } else if (type === 'withdraw') {
      requireThat(r.proposal && r.proposal.id === command.proposalId, 409, '현재 제안과 일치하지 않습니다.');
      requireThat(r.proposal.by === role, 403, '제안한 사람만 철회할 수 있습니다.');
      r.proposal = null; r.lastEvent = '제안을 철회해 기존 승차지점을 유지합니다.';
    } else if (type === 'end') {
      r.outcome = r.stage === 'boarded' ? 'completed' : 'cancelled';
      r.status = 'ended'; r.proposal = null; r.invite = null;
      r.lastEvent = `${role === 'driver' ? '운전자' : '탑승자'}가 ` + (r.outcome === 'completed' ? '탑승 확인 후 이번 픽업을 완료했습니다.' : '탑승 완료 없이 이번 약속을 취소했습니다.');
    } else throw new Fault(400, '알 수 없는 요청입니다.');
    r.requests.add(requestId); r.version++;
    return this.read(id, credential);
  }
}

import http from 'node:http';
import { readFile } from 'node:fs/promises';
import { fileURLToPath } from 'node:url';
import { Board, Fault } from './domain.mjs';
const files = { '/': ['index.html', 'text/html'], '/app.mjs': ['app.mjs', 'text/javascript'], '/style.css': ['style.css', 'text/css'], '/fonts/AstaSans.ttf': ['fonts/AstaSans.ttf', 'font/ttf'], '/fonts/OFL.txt': ['fonts/OFL.txt', 'text/plain'] };
export function createServer(board = new Board()) {
  return http.createServer(async (req, res) => {
    const reply = (status, value) => { res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' }); res.end(JSON.stringify(value)); };
    res.setHeader('Cache-Control', 'no-store'); res.setHeader('X-Content-Type-Options', 'nosniff');
    res.setHeader('Referrer-Policy', 'no-referrer');
    res.setHeader('Content-Security-Policy', "default-src 'self'; script-src 'self'; style-src 'self'; object-src 'none'; frame-ancestors 'none'; base-uri 'none'; form-action 'self'");
    try {
      const origin = `http://127.0.0.1:${res.socket.localPort}`;
      if (req.headers.host !== new URL(origin).host || (req.headers.origin && req.headers.origin !== origin)) throw new Fault(403, '로컬 주소에서만 사용할 수 있습니다.');
      const path = new URL(req.url, origin).pathname;
      if (req.method === 'GET' && files[path]) {
        const [name, mime] = files[path]; const body = await readFile(new URL(name, import.meta.url));
        res.writeHead(200, { 'Content-Type': `${mime}; charset=utf-8` }); res.end(body); return;
      }
      let body = {};
      if (req.method === 'POST') {
        if (req.headers['content-type'] !== 'application/json') throw new Fault(415, 'JSON 요청만 지원합니다.');
        let raw = ''; for await (const chunk of req) { raw += chunk; if (Buffer.byteLength(raw) > 4096) throw new Fault(413, '요청이 너무 큽니다.'); }
        try { body = JSON.parse(raw); } catch { throw new Fault(400, '잘못된 JSON입니다.'); }
        if (!body || Array.isArray(body) || typeof body !== 'object') throw new Fault(400, '요청 객체가 필요합니다.');
      }
      if (path === '/api/rooms' && req.method === 'POST') return reply(201, board.create(body.place));
      const match = path.match(/^\/api\/rooms\/([a-f0-9-]{36})(?:\/(join|actions))?$/);
      if (!match) throw new Fault(404, '찾을 수 없습니다.');
      const [, id, action] = match, credential = req.headers.authorization?.replace(/^Bearer /, '');
      if (req.method === 'GET' && !action) return reply(200, board.read(id, credential));
      if (req.method === 'POST' && action === 'join') return reply(200, board.join(id, body.invite));
      if (req.method === 'POST' && action === 'actions') return reply(200, board.act(id, credential, body));
      throw new Fault(405, '지원하지 않는 요청입니다.');
    } catch (error) { reply(error instanceof Fault ? error.status : 500, { error: error instanceof Fault ? error.message : '서버 오류입니다. 다시 확인해 주세요.' }); }
  });
}
if (process.argv[1] === fileURLToPath(import.meta.url)) {
  const port = Number(process.env.PICKUP_PORT || 4180);
  createServer().listen(port, '127.0.0.1', () => console.log(`픽업 약속판 http://127.0.0.1:${port} · 로컬 시연 / 메모리 저장`));
}
