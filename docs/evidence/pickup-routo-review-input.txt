export function createRoutoSearch({ apiKey = '', usageApproved = false, fetchImpl = fetch, timeoutMs = 5000 } = {}) {

      if (usageApproved !== true) throw new RoutoError('USAGE_PENDING', 'routo 이용 조건 확인이 필요합니다.', 503);
      if (typeof query !== 'string' || !query.trim() || query.trim().length > 80) throw new RoutoError('INVALID_QUERY', '검색어는 1~80자로 입력하세요.', 400);
      const url = new URL(ENDPOINT); url.searchParams.set('key', apiKey); url.searchParams.set('input', query.trim());
      url.searchParams.set('size', '10'); url.searchParams.set('from', '0');
      if (near !== undefined) { const p = coordinate(near); url.searchParams.set('location', `${p.latitude},${p.longitude}`); }

        combined.throwIfAborted();

          if ([401,403].includes(res.status)) throw new RoutoError('KEY_REJECTED', 'routo 키 또는 사용 권한을 확인하세요.', 502);

        }
        // Bounded streaming read; never log a URL containing the server key or raw provider errors.
        const reader = res.body.getReader(); const chunks = []; let length = 0;
        try { for (;;) { const { done, value } = await reader.read(); if (done) break; length += value.byteLength;

        if (timeout.aborted) throw new RoutoError('TIMEOUT', '검색 응답 시간이 초과됐습니다.', 504);

import { coordinate, toMapPoint, RoutoError } from './contracts.mjs';
// Framework-neutral state machine. No room creation or external write occurs here.
export function createSelectionController({ search, checkPoint, now = () => performance.now(), onChange = () => {} }) {
  let state = { phase: 'idle', results: [], anchor: null, selection: null, error: null }, generation = 0, abort, disposed = false;
  const snapshot = () => structuredClone(state);
  const emit = () => { if (!disposed) onChange(snapshot()); };
  const invalidate = () => { generation++; abort?.abort(); abort = new AbortController(); return generation; };
  return {
    snapshot,
    async search(query) {
      if (disposed) return;
      const gen = invalidate(); state = { phase: 'searching', results: [], anchor: null, selection: null, error: null }; emit();
      try { const results = await search({ query, signal: abort.signal }); if (disposed || gen !== generation) return;
        state = { ...state, phase: results.length ? 'results' : 'empty', results }; emit();
      } catch (error) { if (disposed || gen !== generation) return; state = { ...state, phase: 'error', error: error.code || 'SEARCH_FAILED' }; emit(); }
    },
    selectResult(id) {
      if (disposed) return;
      const result = state.results.find(item => item.id === id); if (!result) throw new RoutoError('INVALID_RESULT', '현재 검색 결과를 선택하세요.', 400);
      invalidate(); state = { ...state, phase: 'choose-point', anchor: structuredClone(result), selection: null, error: null }; emit();
      return coordinate(result.center); // Pan only: a search center is never automatically confirmed.
    },
    async choose(value) {
      if (disposed || !state.anchor) return;
      const gen = invalidate(); state = { ...state, phase: 'checking', selection: null, error: null }; emit();
      try {
        const point = coordinate(value), started = now();
        const verdict = await checkPoint(toMapPoint(point), { signal: abort.signal });
        if (disposed || gen !== generation) return;
        // The check adapter must provide a finite validity duration. Cap it at five seconds.
        if (verdict?.selectable !== true || !Number.isFinite(verdict.validForMs) || verdict.validForMs <= 0)
          state = { ...state, phase: 'blocked', error: verdict?.reason || 'POINT_NOT_VERIFIED' };
        else {
          const expiresAt = started + Math.min(5000, verdict.validForMs);
          state = { ...state, phase: now() < expiresAt ? 'review' : 'expired', selection: { point, expiresAt,
            provenance: { provider: 'routo', providerId: state.anchor.providerId, derivedFromSearch: true, method: 'user-pin' } } };
        }
      } catch (error) { if (disposed || gen !== generation) return; state = { ...state, phase: 'error', error: error.code || 'CHECK_FAILED' }; }
      emit();
    },
    confirm() {
      if (disposed || state.phase !== 'review' || !state.selection || now() >= state.selection.expiresAt) {
        if (!disposed && state.phase === 'review') { state = { ...state, phase: 'expired', selection: null }; emit(); }
        throw new RoutoError('NOT_CONFIRMED', '위치를 다시 확인하세요.', 409);
      }
      const selection = structuredClone(state.selection); state = { ...state, phase: 'confirmed' }; emit();
      return { point: toMapPoint(selection.point), provenance: selection.provenance }; // Server MUST revalidate at write time.
    },
    cancel() { if (disposed) return; invalidate(); state = { phase: 'idle', results: [], anchor: null, selection: null, error: null }; emit(); },
    dispose() { disposed = true; invalidate(); state = { phase: 'disposed', results: [], anchor: null, selection: null, error: null }; }
  };
}

import http from 'node:http';

import { requestOrigin } from '../web-origin.mjs';
// Reusable dispatch: parent server provides its validated JSON body and auth/origin boundary.
export function createRoutoRoutes(search) {
  return async ({ path, method, body, signal }) => {
    if (path === '/api/routo/status' && method === 'GET') return search.status();
    if (path === '/api/routo/search' && method === 'POST') return { results: await search.search({ query: body?.query, near: body?.near, signal }) };
    return undefined;
  };
}
// Loopback-only readiness harness; no map/room state writes and no key ever returned.
export function createPreparationServer(options = {}) {
  const provider = createRoutoSearch(options), dispatch = createRoutoRoutes(provider); let active = 0;
  return http.createServer(async (req, res) => {
    res.setHeader('Cache-Control', 'no-store'); res.setHeader('Content-Type', 'application/json; charset=utf-8');
    res.setHeader('X-Content-Type-Options', 'nosniff');
    const reply = (status, value) => { if (!res.destroyed) { res.writeHead(status); res.end(JSON.stringify(value)); } };
    const abort = new AbortController(); res.on('close', () => { if (!res.writableEnded) abort.abort(); });
    let claimed = false;
    try {
      if (!requestOrigin(req.headers, res.socket.localPort, null)) throw new RoutoError('ORIGIN', '허용된 로컬 주소에서만 사용할 수 있습니다.', 403);
      const path = new URL(req.url, 'http://127.0.0.1').pathname;
      if (!['/api/routo/status','/api/routo/search'].includes(path)) throw new RoutoError('NOT_FOUND', '경로를 찾을 수 없습니다.', 404);
      if (req.method !== (path.endsWith('/status') ? 'GET' : 'POST')) throw new RoutoError('METHOD', '지원하지 않는 요청입니다.', 405);
      if (active >= 2) throw new RoutoError('BUSY', '검색 요청 처리 중입니다.', 429);
      active++; claimed = true;
      let body;
      if (req.method === 'POST') {
        if (req.headers['content-type'] !== 'application/json') throw new RoutoError('CONTENT_TYPE', 'JSON 요청이 필요합니다.', 415);
        let size = 0; const chunks = [];
        for await (const chunk of req) { size += chunk.length; if (size > 4096) throw new RoutoError('TOO_LARGE', '검색 요청이 너무 큽니다.', 413); chunks.push(chunk); }
        try { body = JSON.parse(Buffer.concat(chunks).toString('utf8')); } catch { throw new RoutoError('INVALID_JSON', '요청 형식을 확인하세요.', 400); }
      }
      reply(200, await dispatch({ path, method: req.method, body, signal: abort.signal }));
    } catch (error) { const known = error instanceof RoutoError; reply(known ? error.status : 500, { code: known ? error.code : 'INTERNAL', error: known ? error.message : '요청을 처리하지 못했습니다.' }); }
    finally { if (claimed) active--; }
  });
}
