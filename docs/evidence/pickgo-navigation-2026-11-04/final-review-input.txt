 private void unavailable(String fallback){

  String previous=requested==null?null:getPreferences(MODE_PRIVATE).getString(requested.key()+":outcome",null);

 private static boolean local(Uri uri) {
  return uri != null && "http".equals(uri.getScheme()) && "127.0.0.1".equals(uri.getHost())
   && uri.getPort()==4194 && uri.getUserInfo()==null;

 private boolean home() { return web != null && local(Uri.parse(web.getUrl()==null?"":web.getUrl())) && "/".equals(Uri.parse(web.getUrl()).getPath()); }

 private final NaviHelperEventListener listener = new NaviHelperEventListener() {

 private void verified() {

   case ROUND_UNAVAILABLE:return "회차가 변경됐거나 확인할 수 없습니다. 현대 내비 경로는 자동 취소하지 않았으니 확인하세요.";
   case GUIDING:record("GUIDANCE_CONFIRMED");return "현재 현대 내비가 확정 지점으로 안내 중입니다. 목적지 일치를 확인했습니다.";
   case LAST_GUIDING:return "45초 관찰 종료 시점: 현대 내비 목적지 일치·안내 중. 이후 상태는 현대 내비에서 확인하세요.";
   case TIMEOUT:return "안내 확인 시간 초과 · 재전송하지 않았습니다. 현대 내비에서 확인하세요.";
   default:return "현대 내비 상태 확인 중 · 아직 목적지 일치와 안내를 함께 확인하지 못했습니다.";
  }

 private void checkRound(){
  if(checkingRound||requested==null)return;checkingRound=true;
  final RouteGate.Target target=requested;final String token=requestCredential;
  io.execute(()->{try{RouteGate.Target latest=readTarget(target.id(),token);main.post(()->{
   checkingRound=false;if(destroyed||!sent||requested!=target)return;
   roundIntact=RouteGate.samePlace(target,latest);roomChecked=latest.observed();verified();
  });}catch(Exception e){main.post(()->{checkingRound=false;if(destroyed||!sent||requested!=target)return;roundIntact=false;record("ROUND_UNAVAILABLE");verified();});}});
 }
 private final Runnable tick = new Runnable() { public void run() {
  if(destroyed)return;
  if(sent && now()-requestedAt<45000) {
   if(now()-roomChecked>=3000)checkRound();
   try { navi.getDestinationInfo(); navi.getRouteStateInfo(); }
   catch(Exception e) { connected=false;guiding=false; message("내비 연결이 끊겼습니다. 안내 상태를 확인한 뒤 PICKGO 앱을 다시 실행하세요."); }
   verified();
  } else if(sent) {
   outcome=observationMessage(true);
   getPreferences(MODE_PRIVATE).edit().putString(requested.key()+":outcome",outcome).apply();message(outcome);
   sent=false; record("OBSERVATION_FINISHED");
  }
  if(resumed && !starting && !sent) refresh();
  buttons(); main.postDelayed(this,1500);
 }};

 private void refresh() {
  if(reading||!home()){if(!home()){displayed=null;buttons();}return;}
  reading=true; int gen=generation;
  // evaluateJavascript runs in this WebView's main frame only. Nothing is exposed to subframes.
  web.evaluateJavascript("(()=>{try{return sessionStorage.getItem('pickupSession')}catch(e){return null}})()",raw->{
   if(destroyed||gen!=generation||!home()){reading=false;return;}
   try {
    Object decoded=new JSONTokener(raw).nextValue(); if(!(decoded instanceof String))throw new JSONException("No session");
    JSONObject session=new JSONObject((String)decoded); String id=session.getString("id"), token=session.getString("token");
    validateSession(id,token); roomId=id;credential=token;
    io.execute(()->{try{RouteGate.Target t=readTarget(id,token);main.post(()->{
     reading=false;if(destroyed||gen!=generation||!home()||starting||sent)return;
     displayed=t; if(used(t))message(getPreferences(MODE_PRIVATE).getString(t.key()+":outcome","이미 안내를 요청한 지점입니다. 재실행으로 관찰이 끝났을 수 있습니다. 현대 내비에서 확인하세요."));
     else message(String.format(java.util.Locale.ROOT,"확정 픽업 지점 · %.7f, %.7f · 안내 시작을 누르세요.",t.lat(),t.lon()));buttons();
    });}catch(Exception e){main.post(()->{reading=false;if(gen!=generation||destroyed||starting||sent)return;displayed=null;unavailable("지도 지점·상대 참여·장소 합의를 확인하세요. 연결이 없으면 안내할 수 없습니다.");buttons();});}});
   }catch(Exception e){reading=false;displayed=null;unavailable("지도에서 픽업 지점을 확정하고 상대를 초대하세요.");buttons();}
  });
 }
 private static void validateSession(String id,String token) {
  if(!id.matches("[a-f0-9-]{36}")||!token.matches("[A-Za-z0-9_-]{32}"))throw new IllegalArgumentException("Invalid session");
 }
 private RouteGate.Target readTarget(String id,String token) throws Exception {
  validateSession(id,token); long began=now();
  HttpURLConnection c=(HttpURLConnection)new URL(ORIGIN+"/api/rooms/"+id).openConnection();
  activeConnection=c;
  ScheduledFuture<?> deadline=deadlines.schedule(c::disconnect,4000,TimeUnit.MILLISECONDS);
  c.setConnectTimeout(2500);c.setReadTimeout(2500);c.setInstanceFollowRedirects(false);c.setRequestProperty("Authorization","Bearer "+token);c.setUseCaches(false);
  try{
   if(c.getResponseCode()!=200)throw new IOException("Room unavailable");
   ByteArrayOutputStream body=new ByteArrayOutputStream();byte[] chunk=new byte[2048];int length;
   try(InputStream stream=c.getInputStream()){while((length=stream.read(chunk))!=-1){body.write(chunk,0,length);if(body.size()>16384||now()-began>=4000)throw new IOException("Invalid response");}}
   byte[] bytes=body.toByteArray();if(now()-began>=4000)throw new IOException("Response expired");
   JSONObject r=new JSONObject(new String(bytes,StandardCharsets.UTF_8));JSONArray p=r.getJSONArray("mapPoint");
   if(p.length()!=2||!(p.get(0) instanceof Number)||!(p.get(1) instanceof Number)||!id.equals(r.getString("id")))throw new IOException("Invalid point");
   return RouteGate.validate(id,r.getString("role"),r.getString("status"),r.getBoolean("joined"),r.getString("stage"),!r.has("proposal")||!r.isNull("proposal"),r.getLong("version"),r.getLong("placeRevision"),p.getDouble(0),p.getDouble(1),began);
  }finally{deadline.cancel(false);c.disconnect();activeConnection=null;}
 }
 private void begin(){
  if(!start.isEnabled())return;
  final RouteGate.Target expected=displayed;final String id=roomId,token=credential;final int gen=generation;
  starting=true;buttons();message("최신 합의 상태를 확인하고 있습니다…");record("USER_START");
  io.execute(()->{try{RouteGate.Target latest=readTarget(id,token);main.post(()->{
   if(destroyed)return;starting=false;
   if(gen!=generation||!resumed||!home()||!connected||!RouteGate.fresh(expected,now())||!RouteGate.fresh(latest,now())||!RouteGate.same(expected,latest)||used(latest)){
    displayed=null;message("상태가 바뀌었거나 오래되었습니다. 최신 지점을 다시 확인한 뒤 안내 시작을 누르세요.");record("STALE_BLOCKED");buttons();return;
   }
   // Persist before side effect: restart, rapid taps and uncertain SDK returns must not replay.
   if(!getPreferences(MODE_PRIVATE).edit().putBoolean(latest.key(),true).commit()){message("중복 방지 기록 실패 · 안내를 요청하지 않았습니다.");buttons();return;}
   requested=latest;requestCredential=token;requestedAt=now();roomChecked=latest.observed();roundIntact=true;outcome=null;destinationMatches=false;guiding=false;sent=true;
   try{
    navi.requestRoute(new RouteInfo(latest.lat(),latest.lon(),null,null,"PICKGO 픽업 지점","","","",RouteOption.RECOMMENDED,"",false));
    record("REQUEST_ROUTE_SENT");message("현대 내비에 요청했습니다. 실제 목적지와 안내 상태를 확인 중입니다.");
    navi.requestNavigationScreen(NavigationScreenRequest.Foreground.INSTANCE);
   }catch(Exception e){record("REQUEST_UNCERTAIN");message("내비 요청 결과를 확인하지 못했습니다. 중복 요청 없이 현대 내비에서 확인하세요.");}
   buttons();
  });}catch(Exception e){main.post(()->{if(destroyed)return;starting=false;displayed=null;record("REVALIDATION_FAILED");message("최신 상태 확인 실패 · 안내를 요청하지 않았습니다. 연결 후 다시 확인하세요.");buttons();});}});
 }
