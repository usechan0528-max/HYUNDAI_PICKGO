    headers: { ...(body ? { 'Content-Type': 'application/json' } : {}), ...(authenticated && session ? { Authorization: `Bearer ${session.token}` } : {}) }, ...(body ? { body: JSON.stringify(body) } : {}) }); }

// The HTTP service stays on loopback. TLS terminates at an explicitly configured
// reverse proxy which must preserve Host; forwarded headers are never authority.
export function parsePublicOrigin(value) {
  if (value === undefined || value === '') return null;
  let url;
  try { url = new URL(value); } catch { throw new Error('PICKUP_PUBLIC_ORIGIN must be an HTTPS origin.'); }
  if (url.protocol !== 'https:' || url.username || url.password || url.pathname !== '/' || url.search || url.hash) {
    throw new Error('PICKUP_PUBLIC_ORIGIN must be an HTTPS origin without credentials, path, query or fragment.');
  }
  return url.origin;
}

export function requestOrigin(headers, port, publicOrigin) {
  const origin = publicOrigin || `http://127.0.0.1:${port}`;
  if (headers.host !== new URL(origin).host || (headers.origin !== undefined && headers.origin !== origin)) return null;
  if (headers['sec-fetch-site'] === 'cross-site' && headers.origin !== undefined) return null;
  return origin;
}

      if (entry.ids.size < 100) entry.ids.add(consentId);

    }

      entry.grant = { id: consentId, until: at + GRANT_TTL, wall: this.now() };
      return { granted: true, expiresInMs: GRANT_TTL };
    }
    const grant = entry.grant;
    check(grant && grant.id === consentId && at < grant.until, 409, '공유 동의가 만료되거나 철회되었습니다.');
    const f = input.fix;
    check(f && ['latitude','longitude','accuracy','observedAt','ageMs'].every(k => typeof f[k] === 'number' && Number.isFinite(f[k])), 400, '위치 값을 확인할 수 없습니다.');
    check(Math.abs(f.latitude) <= 90 && Math.abs(f.longitude) <= 180 && f.accuracy > 0 && f.accuracy <= 100, 400, '위치 정확도가 부족합니다. 수동 상태 알림을 이용해 주세요.');
    // Device observation epoch is metadata; compare age only within the same device clock.
    // The authenticated client reports age; this is not proof of physical location.
    check(f.ageMs >= 0 && f.ageMs < 15000, 409, '위치 시각을 확인할 수 없습니다. 다시 동의해 주세요.');
    entry.grant = null;
    entry.sample = { id: consentId, until: at + LOCATION_TTL, receivedAt: this.now(), receivedMono: at, ageAtReceipt: f.ageMs,
      fix: { latitude:f.latitude, longitude:f.longitude, accuracy:f.accuracy, observedAt:f.observedAt } };
    return { shared: true, expiresInMs: LOCATION_TTL };
  }
}

  }

  async stop(message='이 기기에서 공유를 중지했습니다.') {
    const old=this.current; ++this.generation; this.clear(); this.current=null; this.phase='idle';

    catch { if(!this.current) this.notify(`${message} 서버 삭제 미확인 · 마지막 전송 후 최대 30초에 만료됩니다.`); }
  }
  async start(snapshot) {
    if(this.current || !this.allowed() || !this.geo) return;
    const generation=++this.generation, consentId=this.id(), started=this.wall(), deadline=this.mono()+15000;
    this.current=consentId; this.phase='grant'; this.notify('동의 확인 중입니다. 아직 위치를 수집하지 않습니다.');
    const live=()=>this.generation===generation && this.current===consentId && this.allowed() && this.mono()<deadline;
    this.timer=this.schedule(()=>{ if(this.generation===generation) void this.stop('위치 확인 시간이 초과되었습니다.'); },15000);

      this.watch=this.geo.watchPosition(async position=>{

        const {latitude,longitude,accuracy}=position.coords, observedAt=position.timestamp;

          await this.stop('정확하거나 새로운 위치를 얻지 못했습니다. 수동 상태 알림을 이용해 주세요.'); return;
        }
        this.clear(); this.phase='sending';
        try {
          await this.send({type:'publish',consentId,fix:{latitude,longitude,accuracy,observedAt,ageMs:this.wall()-observedAt}});
          if(!live()) { if(this.generation===generation) await this.stop(); return; }
          this.phase='shared'; this.notify('이번 회차 운전자에게 1회 공유했습니다. 30초에 조회가 만료되고 다음 정리 주기에 삭제됩니다.');
          this.timer=this.schedule(()=>{ if(this.generation===generation) void this.stop('공유 시간이 끝났습니다.'); },30000);
        } catch (error) { if(this.generation===generation) await this.stop(error.status>=400 && error.status<500?'서버가 위치를 받지 않았습니다. 다시 확인하고 동의해 주세요.':'전송 결과를 확인하지 못했습니다.'); }
      }, async error=>{ if(this.generation===generation) await this.stop(error.code===1?'위치 권한이 거부되었습니다.':error.code===3?'위치 확인 시간이 초과되었습니다.':'위치를 확인할 수 없습니다.'); },{maximumAge:0,timeout:10000,enableHighAccuracy:true});

  }
}
